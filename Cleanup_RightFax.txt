Start-Transcript -Path "C:\RightFax_Uninstall.log" -Append

# --- Stop Services with WMIC Force ---
$rightFaxServices = @(
    "RightFax Client Service",
    "RightFax RPC Server",
    "RightFax WorkServer",
    "RightFax FaxCtrl"
)

foreach ($svc in $rightFaxServices) {
    try {
        $service = Get-Service -Name $svc -ErrorAction SilentlyContinue
        if ($service.Status -eq "Running") {
            Stop-Service -Name $svc -Force -ErrorAction SilentlyContinue
            Start-Sleep -Seconds 3
        }
    } catch { }

    try {
        wmic service where "name='$svc'" call stopservice | Out-Null
        Write-Host "Forced stop via WMIC: $svc"
    } catch {
        Write-Host "WMIC failed for: $svc"
    }
}

# --- Kill lingering RightFax processes (tray icon, services, etc) ---
$processes = @(
    "faxctrl",
    "faxctrl32",
    "rfwin32",
    "rightfaxclient",
    "rfaserv",
    "rfprint"
)

foreach ($proc in $processes) {
    try {
        Get-Process -Name $proc -ErrorAction SilentlyContinue | Stop-Process -Force
        Write-Host "Killed process: $proc"
    } catch {
        Write-Host "Process not found: $proc"
    }
}

# --- Remove Folders ---
$paths = @(
    "C:\Program Files (x86)\RightFax",
    "C:\Program Files\RightFax",
    "$env:APPDATA\Open Text"
)

foreach ($path in $paths) {
    if (Test-Path $path) {
        try {
            Remove-Item -Path $path -Recurse -Force
            Write-Host "Removed folder: $path"
        } catch {
            Write-Host "Failed to remove folder: $path - $_"
        }
    }
}

# --- Remove Registry Keys ---
$registryKeys = @(
    "HKLM:\SOFTWARE\Open Text",
    "HKLM:\SOFTWARE\WOW6432Node\Open Text",
    "HKCU:\Software\Open Text"
)

foreach ($key in $registryKeys) {
    if (Test-Path $key) {
        try {
            Remove-Item -Path $key -Recurse -Force
            Write-Host "Removed registry key: $key"
        } catch {
            Write-Host "Failed to remove registry key: $key - $_"
        }
    }
}

# --- Remove Start Menu Shortcuts ---
$startMenuPaths = @(
    "$env:APPDATA\Microsoft\Windows\Start Menu\Programs\RightFax*",
    "$env:ProgramData\Microsoft\Windows\Start Menu\Programs\RightFax*"
)

foreach ($shortcut in $startMenuPaths) {
    Get-ChildItem -Path $shortcut -Recurse -ErrorAction SilentlyContinue | Remove-Item -Force -Recurse -ErrorAction SilentlyContinue
    Write-Host "Removed Start Menu shortcuts: $shortcut"
}

# --- Remove orphaned uninstall entries from Add/Remove Programs ---
$uninstallRegPaths = @(
    "HKLM:\SOFTWARE\Microsoft\Windows\CurrentVersion\Uninstall",
    "HKLM:\SOFTWARE\WOW6432Node\Microsoft\Windows\CurrentVersion\Uninstall"
)

foreach ($regPath in $uninstallRegPaths) {
    Get-ChildItem $regPath | ForEach-Object {
        $displayName = ($_ | Get-ItemProperty).DisplayName
        if ($displayName -like "*RightFax*") {
            try {
                Remove-Item -Path $_.PsPath -Recurse -Force
                Write-Host "Removed orphaned uninstall entry: $displayName"
            } catch {
                Write-Host "Failed to remove uninstall entry: $displayName - $_"
            }
        }
    }
}
